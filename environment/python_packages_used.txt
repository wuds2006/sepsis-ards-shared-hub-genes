genomepy         # genome / annotation fetch

harmonypy        # batch-effect correction (python side)

scanpy           # single-cell preprocessing, clustering, UMAP, score_genes
matplotlib       # plotting
mpl_toolkits     # 3D / axes helpers
numpy
openmm           # 8.5.1 - MM/GBSA single-point free energies (Supplementary Methods S12)
openpyxl         # supplementary table I/O
pandas
pdbfixer         # structure preparation prior to MM/GBSA
plip             # protein-ligand interaction fingerprints
py3Dmol          # structure visualisation
pyliftover       # genome-build conversion
rdkit            # ligand preparation / cheminformatics
scipy
sklearn          # machine-learning utilities
meeko            # ligand preparation for AutoDock Vina 1.2.5
pymol            # structure rendering (optional, figure panels)
PIL              # image assembly
